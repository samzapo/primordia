cmake_minimum_required(VERSION 3.20)
project(primordia_resident LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Threads REQUIRED)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/profile.json" CACHE PATH "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../build/godot-cpp" godot-cpp)
add_library(primordia_resident SHARED register.cpp)
target_link_libraries(primordia_resident PRIVATE godot-cpp::template_release Threads::Threads)
if(NOT RESIDENT_OUTPUT_DIR)
  message(FATAL_ERROR "Set RESIDENT_OUTPUT_DIR to the generated benchmark project's native directory")
endif()
set_target_properties(primordia_resident PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${RESIDENT_OUTPUT_DIR}")
