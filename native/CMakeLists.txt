cmake_minimum_required(VERSION 3.20)
project(primordia LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/profile.json" CACHE PATH "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../build/godot-cpp" godot-cpp)
# Compile the original library sources with modern toolchains. No upstream
# demo/dependency builds, generated source edits or 32-bit ARM assembly.
set(LIQUIDFUN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../build/liquidfun-original/liquidfun/Box2D")
file(GLOB_RECURSE LIQUIDFUN_SOURCES CONFIGURE_DEPENDS "${LIQUIDFUN_ROOT}/Box2D/*.cpp")
if(NOT LIQUIDFUN_SOURCES)
  message(FATAL_ERROR "Run tools/fetch_liquidfun.py before native configuration")
endif()
add_library(liquidfun_original STATIC ${LIQUIDFUN_SOURCES})
target_include_directories(liquidfun_original PUBLIC "${LIQUIDFUN_ROOT}")
set_target_properties(liquidfun_original PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(primordia SHARED kernels.cpp)
target_link_libraries(primordia PRIVATE godot-cpp::template_release liquidfun_original)
set_target_properties(primordia PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
