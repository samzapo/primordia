cmake_minimum_required(VERSION 3.20)
project(primordia LANGUAGES CXX)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/profile.json" CACHE PATH "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../build/godot-cpp" godot-cpp)
# Compile the original library sources with modern toolchains. No upstream
# demo/dependency builds, generated source edits or 32-bit ARM assembly.
set(LIQUIDFUN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../build/liquidfun-original/liquidfun/Box2D")
file(GLOB_RECURSE LIQUIDFUN_SOURCES CONFIGURE_DEPENDS "${LIQUIDFUN_ROOT}/Box2D/*.cpp")
if(NOT LIQUIDFUN_SOURCES)
  message(FATAL_ERROR "Run tools/fetch_liquidfun.py before native configuration")
endif()
add_library(liquidfun_original STATIC ${LIQUIDFUN_SOURCES})
target_include_directories(liquidfun_original PUBLIC "${LIQUIDFUN_ROOT}")
set_target_properties(liquidfun_original PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(primordia SHARED kernels.cpp)
target_link_libraries(primordia PRIVATE godot-cpp::template_release liquidfun_original Threads::Threads)
set_target_properties(primordia PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

# Explicit compiler experiments; never enable fast-math or host-only ISA flags.
set(PRIMORDIA_NATIVE_OPT "baseline" CACHE STRING "baseline, lto, pgo-generate, pgo-use")
set_property(CACHE PRIMORDIA_NATIVE_OPT PROPERTY STRINGS baseline lto pgo-generate pgo-use)
if(NOT PRIMORDIA_NATIVE_OPT MATCHES "^(baseline|lto|pgo-generate|pgo-use)$")
  message(FATAL_ERROR "Unsupported PRIMORDIA_NATIVE_OPT")
endif()
if(NOT PRIMORDIA_NATIVE_OPT STREQUAL "baseline")
  include(CheckIPOSupported)
  check_ipo_supported(RESULT PRIMORDIA_IPO_OK OUTPUT PRIMORDIA_IPO_ERROR)
  if(NOT PRIMORDIA_IPO_OK)
    message(FATAL_ERROR "IPO unavailable: ${PRIMORDIA_IPO_ERROR}")
  endif()
  set_property(TARGET primordia PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
endif()
if(PRIMORDIA_NATIVE_OPT MATCHES "^pgo-")
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "This explicit PGO recipe currently requires GCC")
  endif()
  set(PRIMORDIA_PROFILE_DIR "${CMAKE_BINARY_DIR}/pgo-data")
  if(PRIMORDIA_NATIVE_OPT STREQUAL "pgo-generate")
    file(MAKE_DIRECTORY "${PRIMORDIA_PROFILE_DIR}")
    target_compile_options(primordia PRIVATE "-fprofile-generate=${PRIMORDIA_PROFILE_DIR}" "-fprofile-update=atomic")
    target_link_options(primordia PRIVATE "-fprofile-generate=${PRIMORDIA_PROFILE_DIR}")
  else()
    file(GLOB_RECURSE PRIMORDIA_PROFILES "${PRIMORDIA_PROFILE_DIR}/*.gcda")
    if(NOT PRIMORDIA_PROFILES)
      message(FATAL_ERROR "PGO requires a representative run of the pgo-generate build first")
    endif()
    target_compile_options(primordia PRIVATE "-fprofile-use=${PRIMORDIA_PROFILE_DIR}" "-Werror=missing-profile" "-Werror=coverage-mismatch")
    target_link_options(primordia PRIVATE "-fprofile-use=${PRIMORDIA_PROFILE_DIR}")
  endif()
endif()

# Small standalone scheduler regression; no engine process is required.
add_executable(primordia_worker_test ../tests/simulation_workers.cpp)
target_link_libraries(primordia_worker_test PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(primordia_worker_test PRIVATE /UNDEBUG)
else()
  target_compile_options(primordia_worker_test PRIVATE -UNDEBUG -fno-exceptions)
endif()
