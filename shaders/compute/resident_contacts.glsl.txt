#[compute]
#version 450
// Exact, stable grid plus immutable circular screening. No capped contact loss.
layout(local_size_x=128) in;
layout(rgba32f,set=0,binding=0) uniform image2D state;
layout(set=0,binding=1,std430) buffer Order { uvec2 order[]; };
layout(set=0,binding=2,std430) buffer Ranges { uvec2 ranges[]; };
layout(set=0,binding=3,std430) buffer Counts { uint counts[]; };
layout(set=0,binding=4,std430) buffer Pairs { uint pairs[]; };
layout(set=0,binding=5,std430) buffer Flags { uint reacted[]; };
layout(set=0,binding=6,std430) buffer Stats { uint stats[]; };
layout(set=0,binding=7,std430) buffer Impulses { vec2 impulses[]; };
layout(push_constant,std430) uniform Params { uint count;uint padded;uint stage;uint level;uint stride;uint mode;float acid;uint unused; } pc;
const vec2 WORLD=vec2(4096,2560);
const uint CAP=64u;
ivec2 pixel(uint i) { return ivec2(int(i%256u)*3,int(i/256u)); }
vec4 pose(uint i) {return imageLoad(state,pixel(i));}
vec4 motion(uint i) {return imageLoad(state,pixel(i)+ivec2(1,0));}
vec4 matter(uint i) {return imageLoad(state,pixel(i)+ivec2(2,0));}
uint bin(ivec2 p) {p=ivec2(mod(vec2(p),vec2(256,160)));return uint(p.y*256+p.x);}
vec2 offset(vec2 a,vec2 b) {return mod(b-a+WORLD*.5,WORLD)-WORLD*.5;}
bool less_key(uvec2 a,uvec2 b) {return a.x<b.x || (a.x==b.x && a.y<b.y);}
void main() {
 uint i=gl_GlobalInvocationID.x;
 if(pc.stage==0u) {
  if(i<40960u)ranges[i]=uvec2(0);
  if(i<pc.count){counts[i]=0u;reacted[i]=0u;}
  if(i<16u)stats[i]=0u;
  if(i<pc.padded) {
   uint cell=0xffffffffu;
   if(i<pc.count && motion(i).w>=0. && motion(i).w!=3.)
    cell=bin(ivec2(floor(mod(pose(i).xy,WORLD)/16.)));
   order[i]=uvec2(cell,i);
  }
  return;
 }
 if(pc.stage==1u) {
  if(i>=pc.padded)return;
  uint j=i^pc.stride;if(j<=i)return;
  uvec2 a=order[i],b=order[j];
  bool ascending=(i&pc.level)==0u;
  if(ascending ? less_key(b,a) : less_key(a,b)){order[i]=b;order[j]=a;}
  return;
 }
 if(pc.stage==2u) {
  if(i>=pc.padded)return;
  uint cell=order[i].x;if(cell==0xffffffffu)return;
  if(i==0u || order[i-1u].x!=cell)ranges[cell].x=i;
  if(i+1u==pc.padded || order[i+1u].x!=cell)ranges[cell].y=i+1u;
  return;
 }
 if(pc.stage==3u) {
  if(i>=pc.count || motion(i).w<0. || motion(i).w==3.)return;
  vec2 at=pose(i).xy;float radius=matter(i).x*.5;
  // Every chemical radius is <=8.5. Retain the same tick-cached radius.
  float reach=radius+8.5+.001;
  ivec2 lo=ivec2(floor((at-reach)/16.)),hi=ivec2(floor((at+reach)/16.));
  uint total=0u,checksum=0u;
  for(int x=lo.x;x<=hi.x;x++)for(int y=lo.y;y<=hi.y;y++) {
   uvec2 range=ranges[bin(ivec2(x,y))];
   for(uint k=range.x;k<range.y;k++) {
    uint j=order[k].y;if(j<=i)continue;
    vec2 d=offset(at,pose(j).xy);float c=radius+matter(j).x*.5;
    if(dot(d,d)>c*c)continue;
    if(total<CAP)pairs[i*CAP+total]=j;
    total++;checksum^=j;
   }
  }
  counts[i]=total;
  // Exact count and checksum can be compared without exporting millions of pairs.
  atomicAdd(stats[0],total);atomicXor(stats[1],checksum);
  if(total>CAP)atomicAdd(stats[2],1u);
 }
}
