#[compute]
#version 450
layout(local_size_x=128) in;
layout(rgba32f,set=0,binding=0) readonly uniform image2D state;
layout(set=0,binding=1,std430) readonly buffer Identity {uvec2 identity[];};
layout(set=0,binding=2,std430) readonly buffer Views {vec4 views[];};
layout(set=0,binding=3,std430) buffer Publication {uvec4 header;vec4 records[];};
layout(set=0,binding=4,std430) readonly buffer Reports {vec4 reports[];};
layout(push_constant,std430) uniform Params {uint count;uint limit;uint views;uint pad;} pc;
const vec2 WORLD=vec2(4096,2560);
ivec2 pixel(uint i){return ivec2(int(i%256u)*3,int(i/256u));}
void main(){
 uint i=gl_GlobalInvocationID.x;if(i>=pc.count)return;
 vec4 motion=imageLoad(state,pixel(i)+ivec2(1,0));if(motion.w<0.)return;
 vec4 pose=imageLoad(state,pixel(i));bool visible=false;
 for(uint v=0u;v<pc.views;v++){
  vec2 d=mod(pose.xy-views[v].xy+WORLD*.5,WORLD)-WORLD*.5;
  if(all(lessThanEqual(abs(d),views[v].zw+vec2(32)))){visible=true;break;}
 }
 if(!visible)return;
 uint out_id=atomicAdd(header.x,1u);if(out_id>=pc.limit)return;
 vec4 chemical=imageLoad(state,pixel(i)+ivec2(2,0));
 records[out_id*2u]=vec4(pose.xy,pose.z,motion.w);
 records[out_id*2u+1u]=vec4(chemical.y,motion.w==3.?reports[i*4u+1u].y:chemical.x,uintBitsToFloat(i),uintBitsToFloat(identity[i].x));
}
