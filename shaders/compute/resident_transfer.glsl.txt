#[compute]
#version 450
layout(local_size_x=128) in;
layout(rgba32f,set=0,binding=0) uniform image2D state;
layout(set=0,binding=1,std430) readonly buffer Metadata {vec4 metadata[];};
layout(set=0,binding=2,std430) readonly buffer Identity {uvec2 identity[];};
layout(set=0,binding=3,std430) buffer Transfer {uvec4 header;vec4 records[];};
layout(set=0,binding=4,std430) readonly buffer Regions {vec4 regions[];};
layout(push_constant,std430) uniform Params {uint count;uint limit;uint pad0;uint pad1;vec4 region;} pc;
const vec2 WORLD=vec2(4096,2560);
ivec2 pixel(uint i){return ivec2(int(i%256u)*3,int(i/256u));}
void main(){
 uint i=gl_GlobalInvocationID.x;if(i>=pc.count)return;
 vec4 motion=imageLoad(state,pixel(i)+ivec2(1,0));if(motion.w<0.)return;
 vec4 pose=imageLoad(state,pixel(i));
 vec2 delta=mod(pose.xy-pc.region.xy+WORLD*.5,WORLD)-WORLD*.5;
 if(pc.pad1==0u){if(all(lessThan(abs(delta),pc.region.zw)))return;}
 else for(uint v=0u;v<pc.pad0;v++){vec2 d=mod(pose.xy-regions[v].xy+WORLD*.5,WORLD)-WORLD*.5;if(all(lessThan(abs(d),regions[v].zw)))return;}
 uint out_id=atomicAdd(header.x,1u);if(out_id>=pc.limit)return;
 uint out_at=out_id*8u;
 for(uint j=0u;j<3u;j++)records[out_at+j]=imageLoad(state,pixel(i)+ivec2(int(j),0));
 for(uint j=0u;j<4u;j++)records[out_at+3u+j]=metadata[i*4u+j];
 records[out_at+7u]=uintBitsToFloat(uvec4(i,identity[i],0));
 // Transfer buffer now owns the complete state. Untransferred overflow remains
 // alive on the GPU. The caller must retain this receipt until CPU acceptance.
 motion.w=-1.;imageStore(state,pixel(i)+ivec2(1,0),motion);
 imageStore(state,pixel(i)+ivec2(2,0),vec4(0));
}
