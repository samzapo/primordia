void emit(int s,double q,vec2 at,int64_t source,int64_t compartment,double fuel,float spread){

  int best=-1;int64_t identity=0l;
  for(int x=low.x;x<=high.x;x++)for(int y=low.y;y<=high.y;y++){ivec2 bin=ivec2(x,y);for(int j=g[bucket(s,bin)];j>=0;j=link[j]){
   if(!in_bin(uint(j),bin)||!near(uint(j),s,at,2.0,-1)||source_of(uint(j))!=source||compartment_of(uint(j))!=compartment||a[j].q.x>=quantum)continue;
   int64_t id=int64_t(uint64_t(a[j].identity.x)|(uint64_t(a[j].identity.y)<<32));if(best<0||id<identity){best=j;identity=id;}
  }}
  if(best<0)break;double added=min(left,quantum-a[best].q.x);a[best].q+=dvec2(added,fuel<0.0?added*potentials[s]:fuel*added/q);c[32+s]+=added;left-=added;
 }

  if(slot>=uint(c[1])){c[6]=1.0;return;}

  uint64_t id=uint64_t(uint(c[2]))|(uint64_t(uint(c[3]))<<32);
