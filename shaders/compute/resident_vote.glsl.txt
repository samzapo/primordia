#[compute]
#version 450
layout(local_size_x=128) in;
layout(rgba32f,set=0,binding=0) uniform image2D state;
layout(set=0,binding=1,std430) readonly buffer Order { uvec2 order[]; };
layout(set=0,binding=2,std430) readonly buffer Ranges { uvec2 ranges[]; };
layout(set=0,binding=3,std430) readonly buffer Counts { uint counts[]; };
layout(set=0,binding=4,std430) readonly buffer Pairs { uint pairs[]; };
layout(set=0,binding=5,std430) buffer Flags { uint reacted[]; };
layout(set=0,binding=6,std430) buffer Stats { uint stats[]; };
layout(set=0,binding=7,std430) buffer Impulses { vec2 impulses[]; };
layout(set=0,binding=8,std430) readonly buffer Acidity { float acid_field[]; };
layout(set=0,binding=9,std430) buffer Events { vec4 events[]; };
layout(push_constant,std430) uniform Params { uint count;uint padded;uint stage;uint level;uint stride;uint mode;float acid;uint tick; } pc;
const vec2 WORLD=vec2(4096,2560);
/*CATALOG_NUTRIENT_BINDING*/
ivec2 pixel(uint i) {return ivec2(int(i%256u)*3,int(i/256u));}
vec4 pose(uint i) {return imageLoad(state,pixel(i));}
vec4 motion(uint i) {return imageLoad(state,pixel(i)+ivec2(1,0));}
vec4 matter(uint i) {return imageLoad(state,pixel(i)+ivec2(2,0));}
uint bin(ivec2 p) {p=ivec2(mod(vec2(p),vec2(256,160)));return uint(p.y*256+p.x);}
vec2 offset(vec2 a,vec2 b) {return mod(b-a+WORLD*.5,WORLD)-WORLD*.5;}

uint acid_index(ivec2 p){p=ivec2(mod(vec2(p),vec2(64,40)));return 1u+uint(p.y*64+p.x);}
float sample_acid(vec2 at){
 if(acid_field[0]<0.)return pc.acid;
 vec2 p=mod(at,WORLD)/64.-.5;ivec2 cell=ivec2(floor(p));vec2 f=fract(p);
 return mix(mix(acid_field[acid_index(cell)],acid_field[acid_index(cell+ivec2(1,0))],f.x),
  mix(acid_field[acid_index(cell+ivec2(0,1))],acid_field[acid_index(cell+ivec2(1,1))],f.x),f.y);
}

uint hash_key(uint v){v^=v>>16;v*=0x7feb352du;v^=v>>15;v*=0x846ca68bu;return v^(v>>16);}
bool eligible(float a,float b,vec4 am,vec4 bm,float acid){
 if((a==6.&&b==2.)||(a==2.&&b==6.))return acid<.9;
 if((a==7.&&(b==4.||b==11.||b==12.))||((a==4.||a==11.||a==12.)&&b==7.))return true;
 if((a==8.&&b==9.)||(a==9.&&b==8.))return min(am.z,bm.z)>0.;
 if(a==2.&&b==1.)return bm.y>=am.y*.15&&acid<.8;
 if(a==1.&&b==2.)return am.y>=bm.y*.15&&acid<.8;
 if((a==0.&&b==4.)||(a==4.&&b==0.))return true;
 return a==b&&(a==0.||a==1.||a==2.||a==5.)&&am.y+bm.y<=(a==0.?nutrient_merge_limit:/*CATALOG_MERGE_LIMIT*/)&&am.y+bm.y>0.;
}
void main(){
 uint i=gl_GlobalInvocationID.x;if(i>=pc.count)return;
 reacted[i]=0xffffffffu;impulses[i]=vec2(0);
 vec4 pv=motion(i),p=pose(i),pm=matter(i);
 if(pv.w<0.||pv.w==3.)return;
 float reach=pm.x*.5+8.5+.001;
 ivec2 lo=ivec2(floor((p.xy-reach)/16.)),hi=ivec2(floor((p.xy+reach)/16.));
 uint best=0u,best_random=0u,best_lo=0u,best_hi=0u;bool found=false;
 vec2 impulse=vec2(0);
 for(int x=lo.x;x<=hi.x;x++)for(int y=lo.y;y<=hi.y;y++){
  uvec2 range=ranges[bin(ivec2(x,y))];
  for(uint k=range.x;k<range.y;k++){
   uint j=order[k].y;if(j==i)continue;
   vec4 b=pose(j),bv=motion(j),bm=matter(j);
   if(bv.w<0.||bv.w==3.)continue;
   // Canonical endpoint order gives exactly the same score at both ends.
   uint low=min(i,j),high=max(i,j);
   vec4 lp=pose(low),hp=pose(high);
   vec2 d=offset(lp.xy,hp.xy);float ds=dot(d,d),c=(pm.x+bm.x)*.5;
   if(ds>c*c)continue;
   float dist=sqrt(ds);vec2 n=ds>.000001?d/dist:vec2(1,0);
   vec2 outward=i==low?n:-n;
   if(ds<c*c){impulse-=outward*(c-dist)*2./float(max(1u,max(counts[pc.count+i],counts[pc.count+j])));if(i==low)atomicAdd(stats[3],1u);}
   float acid=sample_acid(lp.xy);
   if(pc.mode!=3u||!eligible(pv.w,bv.w,pm,bm,acid))continue;
   bool enzyme=(pv.w==6.&&bv.w==2.)||(pv.w==2.&&bv.w==6.)||(pv.w==0.&&bv.w==0.);
   float threshold=.5+(enzyme?.46*pow(clamp(abs(acid-.4)/.4,0.,1.),2.):0.);
   float alignment=min(dot(vec2(cos(lp.z),sin(lp.z)),n),dot(vec2(cos(hp.z),sin(hp.z)),-n));
   if(alignment<threshold)continue;
   // Quantized alignment preference; random ordering only within equal scores.
   uint score=uint(clamp(alignment,0.,1.)*1024.);
   uint random=hash_key(low^hash_key(high)^hash_key(pc.tick));
   bool better=!found||score>best||(score==best&&(random>best_random||
    (random==best_random&&(low>best_lo||(low==best_lo&&high>best_hi)))));
   if(better){found=true;best=score;best_random=random;best_lo=low;best_hi=high;reacted[i]=j;}
  }
 }
 impulses[i]=impulse;
}
