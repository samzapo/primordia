#[compute]
#version 450
layout(local_size_x=128) in;
layout(rgba32f,set=0,binding=0) uniform image2D state;
layout(set=0,binding=1,std430) readonly buffer Identity { uvec2 identity[]; };
layout(set=0,binding=2,std430) readonly buffer Reports { vec4 reports[]; };
layout(set=0,binding=3,std430) readonly buffer Bodies { vec4 bodies[]; };
layout(set=0,binding=4,std430) readonly buffer Points { vec2 points[]; };
layout(set=0,binding=5,std430) readonly buffer Info { uvec4 info[]; };
layout(set=0,binding=6,std430) readonly buffer Decoys { vec4 decoys[]; };
layout(set=0,binding=7,std430) buffer Nominations { uvec2 nomination[]; };
layout(set=0,binding=8,std430) readonly buffer Order { uvec2 order[]; };
layout(set=0,binding=9,std430) readonly buffer Ranges { uvec2 ranges[]; };
layout(set=0,binding=10,std430) buffer Receipt { vec4 receipt[]; };
layout(push_constant,std430) uniform Params {uint count;uint bodies_count;uint stage;uint output_at;vec4 scope;} pc;
const vec2 WORLD=vec2(4096,2560);
ivec2 pixel(uint i){return ivec2(int(i%256u)*3,int(i/256u));}
vec2 offset(vec2 a,vec2 b){return mod(b-a+WORLD*.5,WORLD)-WORLD*.5;}
uint bin(ivec2 p){p=ivec2(mod(vec2(p),vec2(256,160)));return uint(p.y*256+p.x);}
float cross2(vec2 a,vec2 b){return a.x*b.y-a.y*b.x;}
float support(uint body,float angle){
 vec4 shape=bodies[body*4u+1u];uint start=uint(shape.z),n=uint(shape.w);vec2 ray=vec2(cos(angle),sin(angle));
 for(uint k=0u;k<n;k++){
  vec2 a=points[start+k],b=points[start+(k+1u)%n],edge=b-a;float denom=cross2(ray,edge);if(abs(denom)<.000001)continue;
  float t=cross2(a,ray)/denom,d=cross2(a,edge)/denom;
  if(t>=0.&&t<=1.&&d>=0.)return length(mix(a,b,t));
 }
 return bodies[body*4u].w/1.05;
}
uint hash32(uint x){x=((x^(x>>16))*0x7feb352du);x=((x^(x>>15))*0x846ca68bu);return x^(x>>16);}
bool hidden(vec2 p){return pc.scope.z<0.||any(greaterThanEqual(abs(offset(pc.scope.xy,p)),pc.scope.zw+vec2(128)));}
bool encounter(uint slot,uint body,float dt){
 uint key=slot^hash32(identity[slot].x)^hash32(info[body].x)^hash32(uint(bodies[body*4u+3u].z))^0x761abcd1u;
 return (float(hash32(key)>>8)+.5)/16777216.<1.-exp(-2.*dt);
}
void consume(uint slot){
 vec4 v=imageLoad(state,pixel(slot)+ivec2(1,0));v.w=-1.;
 imageStore(state,pixel(slot)+ivec2(1,0),v);imageStore(state,pixel(slot)+ivec2(2,0),vec4(0));
}
void main(){
 uint i=gl_GlobalInvocationID.x;
 if(pc.stage==0u){
  if(i>=pc.count)return;
  nomination[i]=uvec2(0xffffffffu);
  vec4 v=imageLoad(state,pixel(i)+ivec2(1,0));
  if(v.w!=4.&&v.w!=10.)return;
  vec4 p=imageLoad(state,pixel(i));float radius=imageLoad(state,pixel(i)+ivec2(2,0)).x*.5;
  uint lineage=floatBitsToUint(reports[i*4u+3u].y);
  for(uint b=0u;b<pc.bodies_count;b++){
   if(info[b].y==lineage)continue;
   vec4 body=bodies[b*4u];vec2 delta=offset(body.xy,p.xy);
   float coarse_dt=bodies[b*4u+3u].x;
   if(coarse_dt>0.&&(!hidden(p.xy)||!encounter(i,b,coarse_dt)))continue;
   if(v.w==4.){
    float bound=body.w+(coarse_dt>0.?24.:5.);
    if(abs(delta.x)>bound||abs(delta.y)>bound)continue;
    float angle=atan(delta.y,delta.x);
    if(length(delta)<=(coarse_dt>0.?body.w+24.:support(b,angle-body.z)+5.)){nomination[i].x=b;return;}
   }else{
    for(uint d=info[b].z;d<info[b].z+info[b].w;d++){
     vec4 q=decoys[d*2u],meta=decoys[d*2u+1u];vec2 direction=offset(q.xy,p.xy);
     float distance=length(direction);
     if(distance>meta.x+radius||distance<=.000001)continue;
     direction/=distance;
     if(dot(q.zw,direction)<.25||dot(vec2(cos(p.z),sin(p.z)),-direction)<.5)continue;
     nomination[i]=uvec2(b,d);return;
    }
   }
  }
  return;
 }
 // One body per workgroup. Ownership was nominated per particle, so bodies
 // cannot double-consume overlapping toxin/decoy packets.
 if(gl_LocalInvocationID.x!=0u)return;
 uint b=gl_WorkGroupID.x;if(b>=pc.bodies_count)return;
 uint output_at=pc.output_at+b*129u;
 receipt[output_at]=vec4(0);
 for(uint k=1u;k<=128u;k++)receipt[output_at+k]=vec4(0);
 vec4 body=bodies[b*4u];float reach=body.w+(bodies[b*4u+3u].x>0.?24.:5.);
 ivec2 lo=ivec2(floor((body.xy-reach)/16.)),hi=ivec2(floor((body.xy+reach)/16.));
 uint taken=0u,deferred=0u;
 for(int x=lo.x;x<=min(hi.x,lo.x+255);x++)for(int y=lo.y;y<=min(hi.y,lo.y+159);y++){
  uvec2 range=ranges[bin(ivec2(x,y))];
  for(uint k=range.x;k<range.y;k++){
   uint slot=order[k].y;
   if(nomination[slot].x!=b||nomination[slot].y!=0xffffffffu)continue;
   if(imageLoad(state,pixel(slot)+ivec2(1,0)).w!=4.)continue;
   if(taken==128u){deferred++;continue;}
   vec2 delta=offset(body.xy,imageLoad(state,pixel(slot)).xy);
   float amount=imageLoad(state,pixel(slot)+ivec2(2,0)).y;
   receipt[output_at+1u+taken]=vec4(amount,atan(delta.y,delta.x),uintBitsToFloat(slot),uintBitsToFloat(identity[slot].y));
   consume(slot);taken++;
  }
 }
 receipt[output_at]=vec4(taken,deferred,0,0);
 for(uint d=info[b].z;d<info[b].z+info[b].w;d++){
  vec4 q=decoys[d*2u],meta=decoys[d*2u+1u];float radius=meta.x+8.5;
  ivec2 dlo=ivec2(floor((q.xy-radius)/16.)),dhi=ivec2(floor((q.xy+radius)/16.));
  float amount=0.;uint number=0u;
  for(int x=dlo.x;x<=min(dhi.x,dlo.x+255);x++)for(int y=dlo.y;y<=min(dhi.y,dlo.y+159);y++){
   uvec2 range=ranges[bin(ivec2(x,y))];
   for(uint k=range.x;k<range.y;k++){
    uint slot=order[k].y;
    if(nomination[slot]!=uvec2(b,d)||imageLoad(state,pixel(slot)+ivec2(1,0)).w!=10.)continue;
    amount+=imageLoad(state,pixel(slot)+ivec2(2,0)).y;number++;consume(slot);
   }
  }
  receipt[pc.output_at+pc.bodies_count*129u+d]=vec4(number,amount,0,0);
 }
}
