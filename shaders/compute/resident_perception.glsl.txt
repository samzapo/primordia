#[compute]
#version 450
layout(local_size_x=64) in;
layout(rgba32f,set=0,binding=0) uniform image2D state;
layout(set=0,binding=1,std430) buffer Identity { uvec2 identity[]; };
layout(set=0,binding=2,std430) readonly buffer Order { uvec2 order[]; };
layout(set=0,binding=3,std430) readonly buffer Ranges { uvec2 ranges[]; };
layout(set=0,binding=4,std430) buffer Receipts { vec4 receipts[]; };
layout(set=0,binding=5,std430) buffer Reports { vec4 reports[]; };
layout(set=0,binding=6,std430) readonly buffer Products { vec4 products[]; };
layout(set=0,binding=7,std430) readonly buffer Queries { vec4 queries[]; };
layout(set=0,binding=8,std430) buffer Latest { uvec4 latest[]; };
layout(push_constant,std430) uniform Params {
 uint count;uint stage;uint query_count;uint source_count;
 uint emit_start;uint emit_count;uint result_start;float now;
 uint summary_start;uint summaries;uint local_sources;uint pad1;
} pc;
const vec2 WORLD=vec2(4096,2560);
ivec2 pixel(uint i){return ivec2(int(i%256u)*3,int(i/256u));}
vec2 offset(vec2 a,vec2 b){return mod(b-a+WORLD*.5,WORLD)-WORLD*.5;}
uint bin(ivec2 p){p=ivec2(mod(vec2(p),vec2(256,160)));return uint(p.y*256+p.x);}

uint food_hash(ivec2 key){return (uint(key.x)*73856093u^uint(key.y)*19349663u)&127u;}
uint food_bin(uint base,ivec2 key,bool create){
 uint hash=food_hash(key);
 for(uint j=0u;j<128u;j++){
  uint at=base+((hash+j)&127u)*4u;uvec4 stored=latest[at];
  if(stored.z==0u){
   if(!create)return 0xffffffffu;
   latest[at]=uvec4(uint(key.x),uint(key.y),1u,0xffffffffu);
   latest[at+1u]=floatBitsToUint(vec4(0,0,0,3.402823e38));
   latest[at+2u]=uvec4(0,0,0xffffffffu,0);
   return at;
  }
  if(ivec2(stored.xy)==key)return at;
 }
 return 0xffffffffu;
}
void food_query(uint i,vec4 query,uvec4 self){
 uint base=i*1024u,output_at=pc.result_start+i*5u;
 vec4 body=queries[i*4u+3u];float width=max(80.,query.z/4.);
 for(uint j=0u;j<128u;j++)latest[base+j*4u]=uvec4(0);
 ivec2 lo=ivec2(floor((query.xy-query.z)/16.)),hi=ivec2(floor((query.xy+query.z)/16.));
 ivec2 native_lo=ivec2(floor((query.xy-query.z)/64.));
 for(int x=lo.x;x<=min(hi.x,lo.x+255);x++)for(int y=lo.y;y<=min(hi.y,lo.y+159);y++){
  uvec2 range=ranges[bin(ivec2(x,y))];
  for(uint k=range.x;k<range.y;k++){
   uint slot=order[k].y;
   if(imageLoad(state,pixel(slot)+ivec2(1,0)).w!=0.||identity[slot].y==self.x)continue;
   vec2 position=imageLoad(state,pixel(slot)).xy;
   float distance=length(offset(query.xy,position));
   float mass=imageLoad(state,pixel(slot)+ivec2(2,0)).y;
   if(distance>query.z||mass<=0.)continue;
   vec2 delta=offset(body.xy,position);ivec2 key=ivec2(floor(delta/width));
   uint at=food_bin(base,key,true);
   // The radius/4 bins intersect at most 100 distinct wrapped keys. Refuse
   // rather than silently lose food if that invariant is ever changed.
   if(at==0xffffffffu){receipts[output_at+4u].z=3.;if(pc.summaries!=0u)receipts[pc.summary_start+i*385u].z=uintBitsToFloat(1u);return;}
   ivec2 native_bin=ivec2(floor(position/64.));
   ivec2 relative=ivec2(mod(vec2(native_bin-native_lo),vec2(64,40)));
   uint rank=uint(relative.x*40+relative.y)*pc.count+slot;
   latest[at].w=min(latest[at].w,rank);
   vec4 value=uintBitsToFloat(latest[at+1u]);
   value.xyz+=vec3(mass,delta*mass);
   if(distance<value.w||(distance==value.w&&rank<latest[at+2u].z)){
    value.w=distance;latest[at+2u]=uvec4(floatBitsToUint(delta),rank,0);
   }
   latest[at+1u]=floatBitsToUint(value);
  }
 }
 if(pc.summaries!=0u){
  uint summary_at=pc.summary_start+i*385u,n=0u;
  for(uint b=0u;b<128u;b++){
   uint at=base+b*4u;if(latest[at].z==0u)continue;
   for(uint j=0u;j<3u;j++)receipts[summary_at+1u+n*3u+j]=uintBitsToFloat(latest[at+j]);
   n++;
  }
  receipts[summary_at]=uintBitsToFloat(uvec4(n,2u,0u,0u));
 }
 float best=-1.;uint best_rank=0xffffffffu;
 for(uint b=0u;b<128u;b++){
  uint at=base+b*4u;uvec4 entry=latest[at];if(entry.z==0u)continue;
  ivec2 key=ivec2(entry.xy);float mass=0.;vec2 moment=vec2(0),nearest=vec2(0);
  float nearest_distance=3.402823e38;
  for(int x=-1;x<=1;x++)for(int y=-1;y<=1;y++){
   uint neighbor=food_bin(base,key+ivec2(x,y),false);if(neighbor==0xffffffffu)continue;
   vec4 value=uintBitsToFloat(latest[neighbor+1u]);
   mass+=value.x;moment+=value.yz;
   if(value.w<nearest_distance){nearest_distance=value.w;nearest=uintBitsToFloat(latest[neighbor+2u].xy);}
  }
  if(mass<=0.)continue;
  vec2 center=moment/mass;float score=mass/(1.+length(center)/query.z);
  if(score<best||(score==best&&entry.w>=best_rank))continue;
  best=score;best_rank=entry.w;
  bool inner=length(center)<=body.z;
  receipts[output_at]=vec4(body.xy+(inner?nearest:center),mass,inner?1.:0.);
  receipts[output_at+2u].z=uintBitsToFloat(1u);
  receipts[output_at+4u]=vec4(0,score,2,0);
 }
}

void main(){
 uint i=gl_GlobalInvocationID.x;
 if(pc.stage==0u){
  if(i<pc.count)identity[i].y=floatBitsToUint(reports[i*4u+3u].x);
  return;
 }
 if(pc.stage==1u){
  if(i>=pc.count)return;
  vec4 v=imageLoad(state,pixel(i)+ivec2(1,0));
  if(v.w==3.&&reports[i*4u+2u].y<=pc.now){
   v.w=-1.;imageStore(state,pixel(i)+ivec2(1,0),v);
   imageStore(state,pixel(i)+ivec2(2,0),vec4(0));
  }
  return;
 }
 if(pc.stage==2u){
  if(i>=pc.emit_count)return;
  uvec4 receipt=floatBitsToUint(receipts[pc.emit_start+i]);
  if(receipt.x==0u)return;
  for(uint j=0u;j<4u;j++)reports[receipt.y*4u+j]=products[i*4u+j];
  return;
 }
 // Spread small receptor batches across independent workgroups; packing all
 // receptors into one warp serializes divergent food/report work on one SM.
 if(pc.stage==4u){if(gl_LocalInvocationID.x!=0u)return;i=gl_WorkGroupID.x;}
 if(i>=pc.query_count)return;
 // One independent receptor per invocation. Full ordered neighborhood; no
 // contact-list truncation or live-source lookup, and bounded source scratch.
 uint scratch=i*1024u,output_at=pc.result_start+i*5u;
 for(uint j=0u;j<5u;j++)receipts[output_at+j]=vec4(0);
 if(pc.summaries!=0u)receipts[pc.summary_start+i*385u]=vec4(0);
 for(uint s=0u;s<pc.source_count;s++)latest[scratch+s]=uvec4(0);
 vec4 query=queries[i*4u];uvec4 self=floatBitsToUint(queries[i*4u+1u]);
 bool incumbent=floatBitsToUint(queries[i*4u+2u].x)!=0u;
 if(self.w==2u){food_query(i,query,self);return;}
 if(self.w>=3u){
  float best=query.z;uint selected=0xffffffffu;
  ivec2 lo=ivec2(floor((query.xy-query.z)/16.)),hi=ivec2(floor((query.xy+query.z)/16.));
  for(int x=lo.x;x<=min(hi.x,lo.x+255);x++)for(int y=lo.y;y<=min(hi.y,lo.y+159);y++){
   uvec2 range=ranges[bin(ivec2(x,y))];
   for(uint k=range.x;k<range.y;k++){
    uint slot=order[k].y;float kind=imageLoad(state,pixel(slot)+ivec2(1,0)).w;
    if(kind<0.)continue;
    if(self.w==3u&&kind!=2.)continue;
    if(self.w==4u&&((kind!=4.&&kind!=11.&&kind!=12.)||floatBitsToUint(reports[slot*4u+3u].y)==self.y))continue;
    float distance=length(offset(query.xy,imageLoad(state,pixel(slot)).xy));
    if(distance>best)continue;
    float zoom=queries[i*4u+3u].z;
    if(self.w==5u&&zoom>0.){
     float radius=imageLoad(state,pixel(slot)+ivec2(2,0)).x*.5;
     if(distance>max(radius+3./zoom,7./zoom))continue;
    }
    best=distance;selected=slot;
   }
  }
  if(selected!=0xffffffffu){
   if(self.w==5u){
    for(uint j=0u;j<3u;j++)receipts[output_at+j]=imageLoad(state,pixel(selected)+ivec2(int(j),0));
    receipts[output_at+3u]=uintBitsToFloat(uvec4(selected,identity[selected].x,identity[selected].y,0));
    receipts[output_at+4u]=vec4(best,0,4,0);
   }else{
    receipts[output_at+2u].z=uintBitsToFloat(1u);
    receipts[output_at+4u].z=5.;
   }
  }
  return;
 }

 ivec2 lo=ivec2(floor((query.xy-query.z)/16.)),hi=ivec2(floor((query.xy+query.z)/16.));
 ivec2 native_lo=ivec2(floor((query.xy-query.z)/64.));
 for(int x=lo.x;x<=min(hi.x,lo.x+255);x++)for(int y=lo.y;y<=min(hi.y,lo.y+159);y++){
  uvec2 range=ranges[bin(ivec2(x,y))];
  for(uint k=range.x;k<range.y;k++){
   uint slot=order[k].y;
   if(imageLoad(state,pixel(slot)+ivec2(1,0)).w!=3.||identity[slot].y==self.x)continue;
   vec4 life=reports[slot*4u+2u];uvec4 who=floatBitsToUint(reports[slot*4u+1u]);
   if(floatBitsToUint(life.z)==0u||life.y<=pc.now||who.y==self.y)continue;
   float distance=length(offset(query.xy,imageLoad(state,pixel(slot)).xy));
   if(distance>query.z)continue;
   uint source=floatBitsToUint(life.w);
   if(pc.local_sources!=0u){
    source=(who.x*2654435761u)&1023u;
    uint probe=0u;
    for(;probe<1024u;probe++){
     uint held_slot=latest[scratch+source].x;
     if(held_slot==0u||floatBitsToUint(reports[(held_slot-1u)*4u+1u].x)==who.x)break;
     source=(source+1u)&1023u;
    }
    if(probe==1024u){receipts[output_at+4u].z=3.;if(pc.summaries!=0u)receipts[pc.summary_start+i*385u].z=uintBitsToFloat(1u);return;}
   }
   uvec4 held=latest[scratch+source];
   ivec2 native_bin=ivec2(floor(imageLoad(state,pixel(slot)).xy/64.));
   ivec2 relative=ivec2(mod(vec2(native_bin-native_lo),vec2(64,40)));
   uint rank=uint(relative.x*40+relative.y)*pc.count+slot;
   uint first=held.x==0u?rank:min(rank,held.y);
   if(held.x!=0u){
    float old=reports[(held.x-1u)*4u+2u].x;
    if(old>life.x||(old==life.x&&held.w<=rank)){
     latest[scratch+source].y=first;continue;
    }
   }
   latest[scratch+source]=uvec4(slot+1u,first,floatBitsToUint(distance),rank);
  }
 }
 if(pc.summaries!=0u){
  uint summary_at=pc.summary_start+i*385u,n=0u;
  for(uint s=0u;s<pc.source_count;s++){
   uvec4 held=latest[scratch+s];if(held.x==0u)continue;
   if(n>=64u){receipts[summary_at].z=uintBitsToFloat(1u);break;}
   for(uint j=0u;j<4u;j++)receipts[summary_at+1u+n*5u+j]=reports[(held.x-1u)*4u+j];
   receipts[summary_at+5u+n*5u]=vec4(uintBitsToFloat(held.z),0,0,0);n++;
  }
  receipts[summary_at].x=uintBitsToFloat(n);
 }
 float best=-1.;uint best_slot=0u,best_order=0xffffffffu;float best_distance=0.;
 for(uint s=0u;s<pc.source_count;s++){
  uvec4 found=latest[scratch+s];if(found.x==0u)continue;
  uint slot=found.x-1u;vec4 report=reports[slot*4u],life=reports[slot*4u+2u];
  uvec4 who=floatBitsToUint(reports[slot*4u+1u]);
  bool flee=self.w!=0u;
  if(!flee&&report.z>=query.w&&who.w!=0u)continue;
  if(flee&&who.z==0u&&report.z<=query.w)continue;
  float distance=uintBitsToFloat(found.z);
  float score=(flee?100.:2.)+1./(1.+max(0.,pc.now-life.x))+.1/(1.+distance);
  if(flee&&who.z!=0u)score+=1.;
  if(who.x==self.z)score+=.05;
  float weighted=score*((incumbent&&who.x==self.z)?1.2:1.);
  if(weighted<best||(weighted==best&&found.y>=best_order))continue;
  best=weighted;best_slot=slot;best_distance=distance;best_order=found.y;
 }
 if(best>=0.){
  for(uint j=0u;j<4u;j++)receipts[output_at+j]=reports[best_slot*4u+j];
  receipts[output_at+4u]=vec4(best_distance,best,0,0);
 }
}
